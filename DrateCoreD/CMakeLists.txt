cmake_minimum_required(VERSION 3.20)
project(GetRektImGuiKit LANGUAGES CXX)
option(GETREKT_BUNDLED_IMGUI "Build the pinned ImGui 1.93.0 core" ON)
option(GETREKT_BUILD_EXAMPLE "Build standalone Win32/DX11 demo" ON)
option(GETREKT_BUILD_TESTS "Build renderer-free smoke test" ON)
add_library(getrekt_ui INTERFACE)
target_include_directories(getrekt_ui INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(getrekt_ui INTERFACE cxx_std_20)
if(GETREKT_BUNDLED_IMGUI)
  add_library(getrekt_imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp)
  target_include_directories(getrekt_imgui PUBLIC vendor/imgui)
  target_link_libraries(getrekt_ui INTERFACE getrekt_imgui)
endif()
if(GETREKT_BUILD_EXAMPLE AND WIN32)
  add_executable(GetRektUIDemo WIN32 examples/win32_dx11.cpp vendor/imgui/backends/imgui_impl_dx11.cpp vendor/imgui/backends/imgui_impl_win32.cpp)
  target_include_directories(GetRektUIDemo PRIVATE vendor/imgui/backends)
  target_compile_definitions(GetRektUIDemo PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(GetRektUIDemo PRIVATE getrekt_ui d3d11 dxgi d3dcompiler)
endif()
if(GETREKT_BUILD_TESTS)
  enable_testing()
  add_executable(getrekt_smoke tests/smoke.cpp)
  target_link_libraries(getrekt_smoke PRIVATE getrekt_ui)
  add_test(NAME getrekt_smoke COMMAND getrekt_smoke)
  if(TARGET GetRektUIDemo)
    add_test(NAME getrekt_dx11_smoke COMMAND GetRektUIDemo --smoke)
  endif()
endif()
